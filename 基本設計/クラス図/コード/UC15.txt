        -int loginUserId
        -Role role

        +isLoggedIn() boolean

        ADMIN
    }
}

        <<interface>>
        +execute() ReturnHomeResult
    }

    class ReturnHomeService {

        +execute() ReturnHomeResult

        -boolean success
        -Role role
        -String message
    }
}
