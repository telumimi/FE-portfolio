sequenceDiagram
    actor Admin as 管理者
    participant View as ConsoleView
    participant Controller as AdminQuestionController
    participant UseCase as UpdateQuestionUseCase
    participant Service as UpdateQuestionService
    participant QuestionRepo as QuestionRepository
    participant ChoiceRepo as ChoiceRepository
    participant DB as Database

    Admin->>Controller: 問題更新を選択する

    Controller->>QuestionRepo: findAll()
    QuestionRepo->>DB: SELECT * FROM questions
    DB-->>QuestionRepo: 問題一覧
    QuestionRepo-->>Controller: List<Question>

    alt 問題が存在しない
        Controller->>View: showMessage("問題が登録されていません")
        View-->>Admin: 問題未登録メッセージを表示する
        Controller->>View: showAdminMenu()
    else 問題が存在する
        Controller->>View: showQuestions(questions)
        View-->>Admin: 問題一覧を表示する
        Admin->>View: 更新対象の問題を選択する
        View-->>Controller: questionId

        Controller->>QuestionRepo: findById(questionId)
        QuestionRepo->>DB: SELECT * FROM questions WHERE id = ?
        DB-->>QuestionRepo: 問題情報
        QuestionRepo-->>Controller: Question

        Controller->>ChoiceRepo: findByQuestionId(questionId)
        ChoiceRepo->>DB: SELECT * FROM choices WHERE question_id = ?
        DB-->>ChoiceRepo: 選択肢一覧
        ChoiceRepo-->>Controller: List<Choice>

        Controller->>View: showUpdateQuestionForm()
        View-->>Admin: 更新内容の入力を求める
        Admin->>View: 更新内容を入力する
        View-->>Controller: UpdateQuestionCommand

        Controller->>UseCase: execute(command)
        UseCase->>Service: execute(command)

        Service->>Service: 入力内容を検証する

        alt 入力内容が不正
            Service-->>Controller: UpdateQuestionResult(false, "入力内容が不正です")
            Controller->>View: showError(message)
            View-->>Admin: エラー表示・再入力を求める
        else 入力内容が有効
            Service->>QuestionRepo: findById(questionId)
            QuestionRepo->>DB: SELECT * FROM questions WHERE id = ?
            DB-->>QuestionRepo: 問題情報
            QuestionRepo-->>Service: Question

            Service->>Service: Questionを更新する
            Service->>QuestionRepo: update(question)
            QuestionRepo->>DB: UPDATE questions SET ...
            DB-->>QuestionRepo: 更新結果
            QuestionRepo-->>Service: 更新完了

            Service->>Service: Choice一覧を更新する
            Service->>ChoiceRepo: updateAll(choices)
            ChoiceRepo->>DB: UPDATE choices SET ...
            DB-->>ChoiceRepo: 更新結果
            ChoiceRepo-->>Service: 更新完了

            Service-->>Controller: UpdateQuestionResult(true, "問題を更新しました")
            Controller->>View: showMessage(message)
            View-->>Admin: 更新完了メッセージを表示する
            Controller->>View: showAdminMenu()
        end
    end
