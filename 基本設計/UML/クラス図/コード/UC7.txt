namespace Domain {
    class AnswerHistory {
        -int id
        -int userId
        -int questionId
        -int selectedChoiceId
        -boolean correct
    }

    class Question {
        -int id
        -String questionText
        -String explanation
    }

    class Choice {
        -int id
        -String choiceText
        -boolean correct
    }
}

namespace Application {
    class ShowWrongHistoryUseCase {
        <<interface>>
        +execute(ShowWrongHistoryCommand command) ShowWrongHistoryResult
    }

    class ShowWrongHistoryService {
        -AnswerHistoryRepository answerHistoryRepository
        +execute(ShowWrongHistoryCommand command) ShowWrongHistoryResult
    }

    class ShowWrongHistoryCommand {
        -int userId
    }

    class ShowWrongHistoryResult {
        -boolean success
        -String message
        -List~AnswerHistory~ wrongHistories
    }

    class AnswerHistoryRepository {
        <<interface>>
        +findWrongByUserId(int userId) List~AnswerHistory~
    }
}

    class ShowWrongHistoryController {
