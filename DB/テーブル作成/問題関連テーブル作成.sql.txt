CREATE TABLE categories (
    category_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    category_name VARCHAR(100) NOT NULL,
    parent_category_id INT,
    CONSTRAINT fk_categories_parent
        FOREIGN KEY (parent_category_id)
        REFERENCES categories(category_id),
    CONSTRAINT unique_category_name_per_parent
        UNIQUE (parent_category_id, category_name)
);

CREATE TABLE questions (
    question_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    category_id INT NOT NULL,
    question_text TEXT NOT NULL,
    explanation TEXT,
    CONSTRAINT fk_questions_category
        FOREIGN KEY (category_id)
        REFERENCES categories(category_id)
);

CREATE TABLE choices (
    choice_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    question_id INT NOT NULL,
    choice_text VARCHAR(255) NOT NULL,
    choice_order INT NOT NULL,
    is_correct BOOLEAN NOT NULL DEFAULT FALSE,
    CONSTRAINT fk_choices_question
        FOREIGN KEY (question_id)
        REFERENCES questions(question_id),
    CONSTRAINT unique_choice_order_per_question
        UNIQUE (question_id, choice_order)
);

CREATE TABLE answer_histories (
    answer_history_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    user_id INT NOT NULL,
    question_id INT NOT NULL,
    selected_choice_id INT NOT NULL,
    is_correct BOOLEAN NOT NULL,
    CONSTRAINT fk_answer_histories_user
        FOREIGN KEY (user_id)
        REFERENCES users(user_id),
    CONSTRAINT fk_answer_histories_question
        FOREIGN KEY (question_id)
        REFERENCES questions(question_id),
    CONSTRAINT fk_answer_histories_selected_choice
        FOREIGN KEY (selected_choice_id)
        REFERENCES choices(choice_id)
);

CREATE TABLE users (
    user_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    user_name VARCHAR(20) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    role VARCHAR(10) NOT NULL
);
